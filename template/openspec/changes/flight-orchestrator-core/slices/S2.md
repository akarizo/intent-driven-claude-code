# S2 · 状态机核心（纯函数）

## 做什么
新建 `template/plugins/flight/hooks/core.ts`：飞行状态从账本事件推出，下一批动作由纯函数决定。**不得 import `claude-code` 的运行时值**（只允许 `import type`），不调用任何 `$`。S5–S8 都依赖下面的导出，签名按此写死（内部 `State` 结构自定）。

```ts
export type Role = 'executor' | 'reviewer' | 'fixer' | 'resolver'
export type Severity = 'CRITICAL' | 'HIGH' | 'MEDIUM' | 'LOW'
export type Finding = { severity: Severity; file: string; line: number; summary: string; fix: string }
/** slice-gate.py gate / final 打印的 JSON */
export type GateJson = { slice: string; ok: boolean; commit: string; failed: string[]; warnings?: string[]; ceilings?: unknown[]; base?: string; summary?: string }
/** 账本事件（字段表见 spec flight-ledger-events；by = { plugin: 'flight', session }） */
export type FlightEvent = { v: 1; ev: string; change: string; at: string; by: { plugin: 'flight'; session: string }; [k: string]: unknown }
export type Plan = { waves: string[][]; deps: Record<string, string[]> }
/** 一次飞行的上下文（由 S7 在起飞时组装；changeDir 是相对 worktree 根的路径，如 template/openspec/changes/demo） */
export type Flight = { change: string; mainTree: string; changeTree: string; changeDir: string; branch: string; hooksDir: string; model: string; session: string }
export type RunResult = { exitCode: number; stdout: string; stderr: string }
/** 副作用原语：S5 用 $ 实现，S6 / 测试用假实现 */
export interface Io {
  run(argv: readonly string[], opts?: { cwd?: string; timeoutMs?: number; env?: Record<string, string>; stdin?: string }): Promise<RunResult>
  read(path: string): Promise<string | undefined>
  write(path: string, text: string): Promise<void>
  exists(path: string): Promise<boolean>
}
export type Action =
  | { kind: 'dispatch'; role: 'executor'; slice: string; continuation?: { reason: string; failed: string[] } }
  | { kind: 'dispatch'; role: 'resolver'; slice: string; conflicts: string[] }
  | { kind: 'dispatch'; role: 'reviewer'; slice: string; commit: string }
  | { kind: 'dispatch'; role: 'fixer'; findings: Finding[] }
  | { kind: 'gate'; slice: string; agent: string }            // 执行体未经收口结束后的补跑门禁
  | { kind: 'merge'; slice: string; via: 'branch' | 'resolve' | 'fix' }
  | { kind: 'final' }
  | { kind: 'land' }
  | { kind: 'halt'; reason: string }
  | { kind: 'blocked'; slice: string; blockKind: 'gate' | 'infra'; reason: string }

export function reduce(events: readonly FlightEvent[]): State
export function next(state: State, plan: Plan, fpNow: string): Action[]
export function stopVerdict(state: State, agent: string, gate: GateJson): { kind: 'block'; text: string } | { kind: 'accept' }
export function agentOf(state: State, agent: string): { role: Role; slice: string; worktree: string; attempt: number } | undefined
export function currentAttempt(state: State): number          // 无 takeoff 时为 0
export function routingFindings(state: State): Finding[]       // 收口路由对账（HIGH）
export function closeoutLists(state: State): { blocked: { slice: string; kind: 'gate' | 'infra'; reason: string }[]; blocking: Finding[]; deferred: Finding[]; fix: { ok: boolean; commit: string } | null }
export const ev: {   // 事件构造：b = { change, at, session }；产出完整事件（含 v / by）
  takeoff(b, x: { attempt: number; fp: string; branch: string; waves: string[][]; model: string }): FlightEvent
  dispatch(b, x: { attempt: number; slice: string; role: Role; agent: string; model: string; worktree: string }): FlightEvent
  gate(b, x: { attempt: number; agent: string } & GateJson): FlightEvent   // 整份门禁 JSON 都记下（record --json 要用）
  ended(b, x: { attempt: number; agent: string; reason: string; model: string | null }): FlightEvent
  merge(b, x: { attempt: number; slice: string; ok: boolean; commit: string; failed: string[] }): FlightEvent
  review(b, x: { attempt: number; slice: string; agent: string; findings: Finding[] }): FlightEvent
  blocked(b, x: { attempt: number; slice: string; kind: 'gate' | 'infra'; reason: string }): FlightEvent
  final(b, x: { attempt: number; ok: boolean; commit: string; failed: string[] }): FlightEvent
  land(b, x: { attempt: number; verdict: 'ready' | 'draft' }): FlightEvent
  halt(b, x: { attempt: number; reason: string }): FlightEvent
}
```

## 语义（spec flight-state-machine 的展开）
- **attempt**：最新 takeoff 的 attempt。合回（merge ok）与评审（review / 评审阻断）跨 attempt 生效；派发、门禁、阻断只看本 attempt。上次被阻断的切片在本 attempt 重新派发；上次派发了没有结论的切片 → 派发执行体并带 `continuation: { reason: '上一次飞行中断', failed: <它最近一次门禁的 failed 或 []> }`。
- **当前 wave**：第一个含「本 attempt 未结束」切片的 wave（结束 = 任一 attempt 合回，或本 attempt 记了 blocked）。只派发当前 wave 里还没派发过的切片；依赖了本 attempt 已阻断切片的，给 `blocked`（infra，原因 `依赖已 blocked：<S>`）。
- **收口门禁**（agent 结束前的 gate 事件）：一个执行体 / 解冲突 agent 的第 1、2 条红结论 → `stopVerdict` 给 block（text 含 failed 每一项）；第 3 条红 → accept。绿 → accept。
- **结束**（ended 事件）：
  - 执行体：最后一条收口结论绿 → 切片等待合回（`merge via branch`）；最后一条是第 3 条红（已放行）→ `blocked`（gate，原因为该结论的 failed 用「; 」连接）；**没有收口结论，或最后一条红结论之后它被续修了却没再收口** → `gate`（补跑）。补跑结论（ended 之后同一 agent 的 gate 事件）：绿 → 等合回；红且本 attempt 该切片还没续接重派过 → 派发执行体并带 `continuation: { reason: '执行体未正常收口', failed }`；否则 `blocked`（gate）。
  - 解冲突 agent：最后收口结论绿 → `merge via resolve`；否则 `blocked`（infra，原因含「解冲突未通过」），不重派。
  - 评审员：有它的 review 事件 → 该切片评审结束；没有 → `blocked`，slice 为 `review:<S>`，infra，原因含「评审未返回」。
  - 修复 agent：最后收口结论（slice 为 fix 的 gate）绿 → `merge via fix`；否则 `blocked`（slice fix，gate）。
- **合回**：merge ok → 派发该切片评审员（`commit` 取 merge 事件的 commit），此前任一 attempt 已评审过的不再派；merge 不 ok 且 failed 含 `conflict: <文件>` 项 → 派发解冲突 agent（conflicts 为这些文件）；其他失败 → `blocked`（infra）。
- **收尾**：全部 wave 结束、所有已合回切片的评审都结束后：有 CRITICAL / HIGH 且从未有过 ok 的 fix 合回 → 派发修复 agent（findings 只含这些阻断项）；修复结束（合回或阻断）或无阻断项 → 本 attempt 没有 final 时 `final`；final ok → `land`；final 不 ok → `halt`（原因含 failed）。已 land / halt 的 attempt → 空数组。
- **计划指纹**：`fpNow` 与本 attempt 的 takeoff.fp 不同 → 只返回 `halt`（原因含「计划指纹已变」）。
- **同一动作不重复给出**：已派发未结束、已给过的 gate / merge 等，`next` 不再重复。
- **路由对账**：takeoff.model 是别名（opus / sonnet / haiku / fable）。每个 ended 的 `model` 不含该别名或为 null → 一条 HIGH：`file: 'routing'`、`line: 0`、summary 含「路由不符」、角色、期望、实际。
- **closeoutLists**：blocked = 本 attempt 的 blocked 事件；blocking = CRITICAL/HIGH findings（含 routingFindings）；deferred = MEDIUM/LOW；fix = 最近的 fix 合回结果或 null。
