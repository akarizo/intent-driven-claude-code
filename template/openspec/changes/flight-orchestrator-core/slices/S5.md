# S5 · 副作用层与接入空壳

## 做什么
1. 新建 `template/plugins/flight/hooks/io.ts`：`$` 适配层。签名写死（S7 / S8 调用）：
```ts
import type { EngineInterface } from 'claude-code'
import type { Flight, FlightEvent, Io, Role, RunResult } from './core'

export function ioOf($: EngineInterface): Io                    // run → $.process.run（timeoutMs 透传，缺省 30000）；read/write/exists → $.fs
export type Tree = { path: string; branch: string }
export async function trees(io: Io, cwd: string): Promise<Tree[]>  // git -C <cwd> worktree list --porcelain；第一项是主 worktree
export async function judgesDir(io: Io, mainTree: string): Promise<string>  // <main>/.claude/hooks 下有 slice-gate.py 则用它，否则 <main>/template/.claude/hooks；都没有返回 ''
export function judge(io: Io, f: Flight, name: string, args: readonly string[], cwd: string, timeoutMs?: number): Promise<RunResult>
                                                                 // python3 <f.hooksDir>/<name>.py ...args，cwd 为被判定的 worktree
export async function readLedger(io: Io, f: Flight): Promise<{ events: FlightEvent[] } | { error: string }>
                                                                 // judge('ledger', ['show', '--change-dir', <changeTree>/<changeDir>]) 逐行 JSON；非 0 → error 为 stderr
export async function appendEvent(io: Io, f: Flight, event: FlightEvent): Promise<boolean>
                                                                 // cwd = changeTree：hash-object -w --stdin → mktree（100644 blob <sha>\tevent.json）→
                                                                 // rev-parse -q --verify refs/flight/<change>/ledger → commit-tree [-p 父] -m "flight: <ev> <change>" → update-ref <ref> <新> <父或 40 个 0>
                                                                 // update-ref 失败重读链尾再试，至多 3 次；全失败返回 false
export function worktreePath(f: Flight, name: string): string     // <mainTree>/.claude/worktrees/flight-<change>-<name>
export async function ensureWorktree(io: Io, f: Flight, name: string): Promise<{ path: string; created: boolean } | { error: string }>
                                                                 // 已存在 → 原样返回 created:false；否则在 mainTree 执行
                                                                 // git worktree add -b flight/<change>/<name> <path> <f.branch>
export function agentType(role: Role): string                   // executor→flight:executor，reviewer→flight:reviewer，fixer / resolver→flight:fixer
export async function spawnAgent($: EngineInterface, f: Flight, role: Role, cwd: string, prompt: string, description: string): Promise<{ agentId?: string; deny?: string }>
                                                                 // $.agent.spawn({ prompt, description, subagentType: agentType(role), model: f.model, cwd })
export const flights: Map<string, Flight>                       // 本进程内的活跃飞行（S7 起飞时登记）
export async function flightOfAgent(io: Io, agentId: string): Promise<{ flight: Flight; events: FlightEvent[] } | undefined>
                                                                 // 遍历 flights，读各自账本，找 dispatch 事件 agent === agentId 的那一个
```
2. 改 `hooks/register.tsx`，只在 `register` 末尾加两行接入（其余一行不动）：`registerOrchestrator(on)`、`registerLanding(on)`，并在文件头 import。插件只能挂一个 hooks 模块（实测 X11），所以由它接入。
3. 新建两个**空壳**（S7 / S8 会整体替换内容，这里只定接口）：
```ts
// hooks/orchestrator.tsx

// hooks/landing.tsx

```
